from flask import jsonify


def token_required(fn):
    @wraps(fn)
    def wrapper(*args, **kwargs):
        try:
            verify_jwt_in_request()
        except Exception:
            return jsonify({
                "success": False,
                "message": "Authentication required. Please provide a valid access token."
            }), 401

        return fn(*args, **kwargs)

    return wrapper

from functools import wraps

from flask import jsonify
from flask_jwt_extended import get_jwt_identity

from app.extensions import db
from tables.user_table import User


def role_required(*allowed_roles):
    def decorator(fn):
        @wraps(fn)
        def wrapper(*args, **kwargs):
            user_id = get_jwt_identity()
            user = db.session.get(User, int(user_id))

            if user is None:
                return jsonify({
                    "success": False,
                    "message": "User not found."
                }), 404

            if not user.is_active:
                return jsonify({
                    "success": False,
                    "message": "Your account is inactive."
                }), 403

            if user.role not in allowed_roles:
                return jsonify({
                    "success": False,
                    "message": "You do not have permission to access this resource."
                }), 403

            return fn(*args, **kwargs)

        return wrapper
    return decorator

from functools import wraps

from flask import jsonify, g
from flask_jwt_extended import (
    verify_jwt_in_request,
    get_jwt_identity
)

from app.extensions import db
from tables.vendor_table import Vendor


def vendor_required(fn):

    @wraps(fn)
    def wrapper(*args, **kwargs):

        # Verify JWT token
        try:
            verify_jwt_in_request()
            identity = get_jwt_identity()

        except Exception:
            return jsonify({
                "success": False,
                "message": (
                    "Authentication required. "
                    "Please provide a valid token."
                )
            }), 401

        # Verify vendor identity
        if (
            not isinstance(identity, str)
            or not identity.startswith("vendor:")
        ):
            return jsonify({
                "success": False,
                "message": "Vendor authentication required."
            }), 403

        # Extract vendor ID
        try:
            vendor_id = int(identity.split(":", 1)[1])

            if vendor_id <= 0:
                raise ValueError("Invalid vendor ID")

        except (ValueError, IndexError):
            return jsonify({
                "success": False,
                "message": "Invalid vendor identity."
            }), 401

        # Get vendor from database
        vendor = db.session.get(Vendor, vendor_id)

        if vendor is None:
            return jsonify({
                "success": False,
                "message": "Vendor account not found."
            }), 404

        # Check vendor status
        if not vendor.is_active:
            return jsonify({
                "success": False,
                "message": "Your vendor account is inactive."
            }), 403

        if not vendor.is_approved:
            return jsonify({
                "success": False,
                "message": "Your vendor account is awaiting approval."
            }), 403

        # Make vendor ID available to routes
        g.vendor_id = vendor.id
        request_vendor_id = vendor.id

        from flask import request
        request.vendor_id = request_vendor_id

        return fn(*args, **kwargs)

    return wrapper

import logging

from flask import current_app

from alembic import context

config = context.config

fileConfig(config.config_file_name)

def get_engine():
    try:
        # this works with Flask-SQLAlchemy<3 and Alchemical
        return current_app.extensions['migrate'].db.get_engine()
    except (TypeError, AttributeError):
        # this works with Flask-SQLAlchemy>=3
        return current_app.extensions['migrate'].db.engine


def get_engine_url():
    try:
        return get_engine().url.render_as_string(hide_password=False).replace(
            '%', '%%')
    except AttributeError:
        return str(get_engine().url).replace('%', '%%')


# add your model's MetaData object here
# for 'autogenerate' support
# from myapp import mymodel
# target_metadata = mymodel.Base.metadata
config.set_main_option('sqlalchemy.url', get_engine_url())
target_db = current_app.extensions['migrate'].db

# other values from the config, defined by the needs of env.py,
# can be acquired:
# my_important_option = config.get_main_option("my_important_option")
# ... etc.


def get_metadata():
    if hasattr(target_db, 'metadatas'):
        return target_db.metadatas[None]
    return target_db.metadata


def run_migrations_offline():
    """Run migrations in 'offline' mode.

    This configures the context with just a URL
    and not an Engine, though an Engine is acceptable
    here as well.  By skipping the Engine creation
    we don't even need a DBAPI to be available.

    Calls to context.execute() here emit the given string to the
    script output.

    """
    url = config.get_main_option("sqlalchemy.url")
    context.configure(
        url=url, target_metadata=get_metadata(), literal_binds=True
    )

    with context.begin_transaction():
        context.run_migrations()


def run_migrations_online():
    """Run migrations in 'online' mode.

    In this scenario we need to create an Engine
    and associate a connection with the context.

    """

    # this callback is used to prevent an auto-migration from being generated
    # when there are no changes to the schema
    # reference: http://alembic.zzzcomputing.com/en/latest/cookbook.html
    def process_revision_directives(context, revision, directives):
        if getattr(config.cmd_opts, 'autogenerate', False):
            script = directives[0]
            if script.upgrade_ops.is_empty():
                directives[:] = []
                logger.info('No changes in schema detected.')

    conf_args = current_app.extensions['migrate'].configure_args
    if conf_args.get("process_revision_directives") is None:
        conf_args["process_revision_directives"] = process_revision_directives

    connectable = get_engine()

    with connectable.connect() as connection:
        context.configure(
            connection=connection,
            target_metadata=get_metadata(),
            **conf_args
        )

        with context.begin_transaction():
            context.run_migrations()


if context.is_offline_mode():
    run_migrations_offline()
else:
    run_migrations_online()

ALLOWED_ADDRESS_TYPES = {"home", "work", "other"}

REQUIRED_FIELDS = (
    "full_name",
    "phone_number",
    "address_line1",
    "city",
    "state",
    "pincode",
)

def _error(message, status_code):
    return {
        "success": False,
        "message": message,
    }, status_code


def _validate_address_data(data, partial=False):
    """Validate address fields and return cleaned data."""

    if not isinstance(data, dict):
        return None, "Request body must be a JSON object."

    allowed_fields = {
        "full_name",
        "phone_number",
        "address_line1",
        "address_line2",
        "landmark",
        "city",
        "state",
        "pincode",
        "country",
        "address_type",
        "is_default",
    }

    unknown_fields = set(data.keys()) - allowed_fields

    if unknown_fields:
        return None, (
            "Unknown fields: "
            + ", ".join(sorted(unknown_fields))
        )

    if not partial:
        missing_fields = [
            field
            for field in REQUIRED_FIELDS
            if field not in data
        ]

        if missing_fields:
            return None, (
                "Missing required fields: "
                + ", ".join(missing_fields)
            )

    cleaned = {}

    string_limits = {
        "full_name": 100,
        "phone_number": 20,
        "address_line1": 255,
        "address_line2": 255,
        "landmark": 150,
        "city": 100,
        "state": 100,
        "pincode": 10,
        "country": 100,
        "address_type": 20,
    }

    for field, max_length in string_limits.items():
        if field not in data:
            continue

        value = data[field]

        if value is None and field in {
            "address_line2",
            "landmark",
        }:
            cleaned[field] = None
            continue

        if not isinstance(value, str):
            return None, f"{field} must be a string."

        value = value.strip()

        if field in REQUIRED_FIELDS and not value:
            return None, f"{field} is required."

        if field == "address_type":
            value = value.lower()

            if value not in ALLOWED_ADDRESS_TYPES:
                return None, (
                    "address_type must be home, work, or other."
                )

        if len(value) > max_length:
            return None, (
                f"{field} must not exceed {max_length} characters."
            )

        cleaned[field] = value or None

    if "is_default" in data:
        if not isinstance(data["is_default"], bool):
            return None, "is_default must be a boolean."

        cleaned["is_default"] = data["is_default"]

    if "country" not in cleaned and not partial:
        cleaned["country"] = "India"

    if "address_type" not in cleaned and not partial:
        cleaned["address_type"] = "home"

    return cleaned, None


def _set_default_address(user_id, address):
    """Make one address default for the customer."""

    addresses = db.session.execute(
        select(Address).where(Address.user_id == user_id)
    ).scalars().all()

    for item in addresses:
        item.is_default = item.id == address.id

    address.is_default = True

def create_address(user_id, data):
    """Create a delivery address for the logged-in customer."""

    cleaned, error = _validate_address_data(data)

    if error:
        return _error(error, 400)

    try:
        address = Address(
            user_id=user_id,
            **{
                key: value
                for key, value in cleaned.items()
                if key != "is_default"
            },
        )

        db.session.add(address)
        db.session.flush()

def _error(message, status_code):
    return {
        "success": False,

    }, status_code


def _parse_datetime(value, field_name):
    """
    Accept ISO-8601 datetime strings.
    Example: 2026-10-20T10:00:00+05:30
    """
    if value is None or value == "":
        return None, None

    if not isinstance(value, str):
        return None, f"{field_name} must be an ISO-8601 datetime string."

    try:
        parsed = datetime.fromisoformat(value.replace("Z", "+00:00"))

        # Treat a datetime without timezone as UTC.
        if parsed.tzinfo is None:
            parsed = parsed.replace(tzinfo=timezone.utc)

        return parsed, None

    except ValueError:
        return None, f"{field_name} must be a valid ISO-8601 datetime."


def _validate_text(data, field_name, required=False, max_length=None):
    value = data.get(field_name)

    if value is None:
        if required:
            return None, f"{field_name} is required."
        return None, None

    if not isinstance(value, str):
        return None, f"{field_name} must be a string."

    value = value.strip()

    if required and not value:
        return None, f"{field_name} cannot be empty."

    if max_length and len(value) > max_length:
        return None, f"{field_name} must not exceed {max_length} characters."

    return value or None, None


def _validate_order(value):
    if isinstance(value, bool) or not isinstance(value, int):
        return None, "display_order must be an integer."

    if value < 0:
        return None, "display_order cannot be negative."

    return value, None


def _validate_boolean(value, field_name):
    if not isinstance(value, bool):
        return None, f"{field_name} must be true or false."

    return value, None


def _validate_schedule(starts_at, ends_at):
    if starts_at and ends_at and ends_at <= starts_at:
        return "ends_at must be later than starts_at."

    return None


def _is_banner_currently_visible(banner, now=None):
    """
    Determine whether a banner should be shown to customers.
    This does not change the stored is_active value.
    """
    now = now or datetime.now(timezone.utc)

    if not banner.is_active:
        return False

    starts_at = banner.starts_at
    ends_at = banner.ends_at

    if starts_at:
        if starts_at.tzinfo is None:
            starts_at = starts_at.replace(tzinfo=timezone.utc)

        if starts_at > now:
            return False

    if ends_at:
        if ends_at.tzinfo is None:
            ends_at = ends_at.replace(tzinfo=timezone.utc)

        if ends_at <= now:
            return False

    return True

    subtitle, error = _validate_text(

    )
    if error:

    redirect_url, error = _validate_text(

    )
    if error:
        return _error(error, 400)

    position = data.get("position", "home")
    if not isinstance(position, str) or not position.strip():
        return _error("position must be a non-empty string.", 400)

    position = position.strip()
    if len(position) > 50:
        return _error("position must not exceed 50 characters.", 400)

    display_order, error = _validate_order(

    )

        return _error(error, 400)

    schedule_error = _validate_schedule(starts_at, ends_at)

        return _error(schedule_error, 400)

        position=position,
        display_order=display_order,

        ends_at=ends_at,
        is_active=is_active
    )

    try:
        db.session.add(banner)
        db.session.commit()

        return {
            "success": True,
            "message": "Banner created successfully.",

        }, 201

                return _error(error, 400)

        if "is_active" in data:
            active, error = _validate_boolean(data["is_active"], "is_active")
            if error:
                return _error(error, 400)

        for field in ("starts_at", "ends_at"):

def _get_product_price(product):
    """Return the effective product price from the database."""
    if product.discount_price is not None:
        return Decimal(str(product.discount_price))

    return Decimal(str(product.price))


def _get_product_image(product):
    """Return the first product image URL when available."""
    try:
        images = product.images
        if images:
            image = images[0]

            if isinstance(image, dict):
                return image.get("image_url") or image.get("url")

            return (
                getattr(image, "image_url", None)
                or getattr(image, "image_path", None)
                or getattr(image, "url", None)
            )
    except (AttributeError, TypeError):
        pass

    return None


def _serialize_cart(cart):
    """Build a cart response containing product details and totals."""
    items = []
    subtotal = Decimal("0.00")

    for item in cart.items:
        product = item.product

        if product is None:
            continue

        unit_price = _get_product_price(product)
        line_total = unit_price * item.quantity
        subtotal += line_total

        items.append({
            "cart_item_id": item.id,
            "product_id": product.id,
            "product_name": product.product_name,
            "sku": product.sku,
            "image_url": _get_product_image(product),
            "quantity": item.quantity,
            "unit_price": str(unit_price),
            "line_total": str(line_total),
            "available_stock": product.stock_quantity,
            "is_active": product.is_active,
        })

    return {
        "cart_id": cart.id,
        "user_id": cart.user_id,
        "items": items,
        "item_count": len(items),
        "total_quantity": sum(item["quantity"] for item in items),

def _positive_id(value, field):
    if isinstance(value, bool):
        raise ValueError(f"{field} must be a positive integer.")
    try:
        parsed = int(value)
    except (TypeError, ValueError):
        raise ValueError(f"{field} must be a positive integer.")

        raise ValueError(f"{field} must be a positive integer.")

def _decimal(value, field):
    try:
        number = Decimal(str(value))
    except (InvalidOperation, ValueError, TypeError):
        raise ValueError(f"{field} must be a valid number.")
    if not number.is_finite():
        raise ValueError(f"{field} must be finite.")
    return number


def _calculate(sale_amount, commission_rate):
    commission_amount = (sale_amount * commission_rate / Decimal("100")).quantize(
        CENT, rounding=ROUND_HALF_UP
    )
    vendor_amount = (sale_amount - commission_amount).quantize(
        CENT, rounding=ROUND_HALF_UP
    )
    return commission_amount, vendor_amount

def _parse_datetime(value):
    if value is None or value == "":
        return None

    if not isinstance(value, str):
        raise ValueError("Date must be an ISO-8601 string.")

    try:
        result = datetime.fromisoformat(value.replace("Z", "+00:00"))
        if result.tzinfo is None:
            result = result.replace(tzinfo=timezone.utc)
        return result
    except ValueError:
        raise ValueError(
            "Invalid date format. Use ISO-8601, e.g. 2026-12-31T23:59:00Z."
        )


def _parse_decimal(value, field, required=False):
    if value is None or value == "":
        if required:
            raise ValueError(f"{field} is required.")
        return None

    try:
        number = Decimal(str(value))
    except (InvalidOperation, ValueError):
        raise ValueError(f"{field} must be a valid number.")

    if not number.is_finite():
        raise ValueError(f"{field} must be a finite number.")

    return number


def _validate_payload(data, creating=True):
    if not isinstance(data, dict):
        raise ValueError("Request body must be a JSON object.")

    allowed = {
        "code",
        "description",
        "discount_type",
        "discount_value",
        "minimum_order_amount",
        "maximum_discount_amount",
        "usage_limit",
        "starts_at",
        "expires_at",
        "is_active",
    }

    unknown = set(data) - allowed
    if unknown:
        raise ValueError(f"Unknown fields: {', '.join(sorted(unknown))}")

    if creating:
        required = {"code", "discount_type", "discount_value"}
        missing = required - set(data)
        if missing:
            raise ValueError(f"Required fields missing: {', '.join(sorted(missing))}")

    result = {}

    if "code" in data:
        code = data["code"]
        if not isinstance(code, str) or not code.strip():
            raise ValueError("code must be a non-empty string.")
        result["code"] = code.strip().upper()

    if "description" in data:
        description = data["description"]
        if description is not None and not isinstance(description, str):
            raise ValueError("description must be a string or null.")
        result["description"] = description.strip() if description else None

    if "discount_type" in data:
        discount_type = data["discount_type"]
        if discount_type not in ("percentage", "fixed"):
            raise ValueError("discount_type must be 'percentage' or 'fixed'.")
        result["discount_type"] = discount_type

    if "discount_value" in data:
        value = _parse_decimal(data["discount_value"], "discount_value", True)
        if value <= 0:
            raise ValueError("discount_value must be greater than zero.")
        result["discount_value"] = value

    if "minimum_order_amount" in data:
        value = _parse_decimal(
            data["minimum_order_amount"], "minimum_order_amount"
        )
        if value is None or value < 0:
            raise ValueError("minimum_order_amount must be zero or greater.")
        result["minimum_order_amount"] = value

    if "maximum_discount_amount" in data:
        value = _parse_decimal(
            data["maximum_discount_amount"], "maximum_discount_amount"
        )
        if value is not None and value < 0:
            raise ValueError("maximum_discount_amount cannot be negative.")
        result["maximum_discount_amount"] = value

    if "usage_limit" in data:
        value = data["usage_limit"]
        if value is not None:
            if isinstance(value, bool) or not isinstance(value, int) or value <= 0:
                raise ValueError("usage_limit must be a positive integer or null.")
        result["usage_limit"] = value

    if "starts_at" in data:
        result["starts_at"] = _parse_datetime(data["starts_at"])

    if "expires_at" in data:
        result["expires_at"] = _parse_datetime(data["expires_at"])

    if "is_active" in data:
        if not isinstance(data["is_active"], bool):
            raise ValueError("is_active must be true or false.")
        result["is_active"] = data["is_active"]

    final_type = result.get("discount_type")
    final_value = result.get("discount_value")

    if not creating:
        coupon = data.get("_existing_coupon")
        if coupon:
            final_type = final_type or coupon.discount_type
            final_value = final_value or coupon.discount_value

    if final_type == "percentage" and final_value is not None:
        if final_value > 100:
            raise ValueError("Percentage discount cannot exceed 100.")

    start = result.get("starts_at")
    expiry = result.get("expires_at")
    if start and expiry and expiry <= start:
        raise ValueError("expires_at must be later than starts_at.")

    return result

from sqlalchemy import select
from sqlalchemy.exc import IntegrityError, SQLAlchemyError

from app.extensions import db
from tables.inventory_table import Inventory
from tables.product_table import Product
from tables.store_table import Store


ALLOWED_MOVEMENTS = {
    "purchase",
    "sale",
    "return",
    "adjustment",
    "damaged",
}


def record_inventory_movement(
    product_id,
    quantity_change,
    movement_type,
    store_id=None,
    reference_type=None,
    reference_id=None,
    note=None,
    created_by_type="system",
    created_by_id=None,
    commit=True,
):
    """
    Record a stock movement and update the product's current stock.

    Positive quantity_change adds stock.
    Negative quantity_change removes stock.

    If commit=False, the caller owns the transaction.
    """

    try:
        if isinstance(product_id, bool) or isinstance(quantity_change, bool):
            return {
                "success": False,
                "message": "Invalid product or quantity."
            }, 400

        product_id = int(product_id)
        quantity_change = int(quantity_change)

        if product_id <= 0 or quantity_change == 0:
            return {
                "success": False,
                "message": "Product ID must be positive and quantity cannot be zero."
            }, 400

        if movement_type not in ALLOWED_MOVEMENTS:
            return {
                "success": False,
                "message": "Invalid inventory movement type."
            }, 400

        if store_id is not None:
            if isinstance(store_id, bool):
                return {
                    "success": False,
                    "message": "Invalid store ID."
                }, 400

            store_id = int(store_id)

        if reference_id is not None:
            if isinstance(reference_id, bool):
                return {
                    "success": False,
                    "message": "Invalid reference ID."
                }, 400

            reference_id = int(reference_id)

        if created_by_id is not None:
            if isinstance(created_by_id, bool):
                return {
                    "success": False,
                    "message": "Invalid creator ID."
                }, 400

            created_by_id = int(created_by_id)

        if created_by_type not in {"admin", "vendor", "system"}:
            return {
                "success": False,
                "message": "Invalid creator type."
            }, 400

        if note is not None:
            if not isinstance(note, str):
                return {
                    "success": False,
                    "message": "Note must be a string."
                }, 400

            note = note.strip()

            if len(note) > 500:
                return {
                    "success": False,
                    "message": "Note cannot exceed 500 characters."
                }, 400

        # Lock the product row while changing stock.
        product = db.session.execute(
            select(Product)
            .where(Product.id == product_id)
            .with_for_update()
        ).scalar_one_or_none()

        if product is None:
            return {
                "success": False,
                "message": "Product not found."
            }, 404

        if store_id is not None and product.store_id != store_id:
            return {
                "success": False,
                "message": "Product does not belong to this store."
            }, 403

        actual_store_id = product.store_id

        # Verify that the product's store exists.
        store = db.session.get(Store, actual_store_id)

        if store is None:
            return {
                "success": False,
                "message": "Product store not found."
            }, 404

        previous_stock = product.stock_quantity
        new_stock = previous_stock + quantity_change

        if new_stock < 0:
            return {
                "success": False,
                "message": "Insufficient stock."
            }, 400

        inventory = Inventory(
            product_id=product.id,
            store_id=actual_store_id,
            movement_type=movement_type,
            quantity_change=quantity_change,
            previous_stock=previous_stock,
            new_stock=new_stock,
            reference_type=reference_type,
            reference_id=reference_id,
            note=note or None,
            created_by_type=created_by_type,
            created_by_id=created_by_id,
        )

        product.stock_quantity = new_stock
        db.session.add(inventory)

        # Flush so database constraint failures occur before success.
        db.session.flush()

        if commit:
            db.session.commit()

        return {
            "success": True,
            "message": "Inventory movement recorded successfully.",
            "data": inventory.to_dict(),
        }, 201

    except (ValueError, TypeError, OverflowError):
        if commit:
            db.session.rollback()

        return {
            "success": False,
            "message": "Invalid inventory data."
        }, 400

    except IntegrityError:
        if commit:
            db.session.rollback()

        return {
            "success": False,
            "message": "Inventory data violates a database constraint."
        }, 409

    except SQLAlchemyError:
        if commit:
            db.session.rollback()

        return {
            "success": False,
            "message": "Database error while recording inventory movement."
        }, 500


def get_inventory_history(
    vendor_id=None,
    store_id=None,
    product_id=None,
    movement_type=None,
    limit=50,
    offset=0,
):
    """
    Retrieve inventory history.

    Pass vendor_id for vendor-scoped access.
    Admin callers may omit vendor_id.
    """

    try:
        if vendor_id is not None:
            if isinstance(vendor_id, bool):
                return {
                    "success": False,
                    "message": "Invalid vendor ID."
                }, 400

            vendor_id = int(vendor_id)

            if vendor_id <= 0:
                return {
                    "success": False,
                    "message": "Invalid vendor ID."
                }, 400

        if store_id is not None:
            if isinstance(store_id, bool):
                return {
                    "success": False,
                    "message": "Invalid store ID."
                }, 400

            store_id = int(store_id)

            if store_id <= 0:
                return {
                    "success": False,
                    "message": "Invalid store ID."
                }, 400

        if product_id is not None:
            if isinstance(product_id, bool):
                return {
                    "success": False,
                    "message": "Invalid product ID."
                }, 400

            product_id = int(product_id)

            if product_id <= 0:
                return {
                    "success": False,
                    "message": "Invalid product ID."
                }, 400

        if movement_type is not None and movement_type not in ALLOWED_MOVEMENTS:
            return {
                "success": False,
                "message": "Invalid inventory movement type."
            }, 400

        if isinstance(limit, bool) or isinstance(offset, bool):
            return {
                "success": False,
                "message": "Invalid pagination values."
            }, 400

        limit = int(limit)
        offset = int(offset)

        if limit < 1 or limit > 100 or offset < 0:
            return {
                "success": False,
                "message": "Limit must be 1-100 and offset cannot be negative."
            }, 400

        query = select(Inventory)

        if vendor_id is not None:
            query = query.join(
                Store,
                Inventory.store_id == Store.id
            ).where(
                Store.vendor_id == vendor_id
            )

        if store_id is not None:
            query = query.where(Inventory.store_id == store_id)

        if product_id is not None:
            query = query.where(Inventory.product_id == product_id)

        if movement_type is not None:
            query = query.where(
                Inventory.movement_type == movement_type
            )

        count_query = select(db.func.count()).select_from(
            query.order_by(None).subquery()
        )

        total = db.session.execute(count_query).scalar_one()

        records = db.session.execute(
            query
            .order_by(Inventory.created_at.desc(), Inventory.id.desc())
            .limit(limit)
            .offset(offset)
        ).scalars().all()

        return {
            "success": True,
            "message": "Inventory history retrieved successfully.",
            "count": len(records),
            "total": total,
            "limit": limit,
            "offset": offset,
            "data": [record.to_dict() for record in records],
        }, 200

    except (ValueError, TypeError, OverflowError):
        return {
            "success": False,
            "message": "Invalid inventory filters."
        }, 400

    except SQLAlchemyError:
        return {
            "success": False,
            "message": "Database error while retrieving inventory history."
        }, 500

def _get_product_price(product):
    """Return the effective selling price."""
    if product.discount_price is not None:
        return Decimal(str(product.discount_price))

    return Decimal(str(product.price))


def _get_product_image(product):
    """Return the first product image URL when available."""
    try:
        images = product.images

        if images:
            image = images[0]

            if isinstance(image, dict):
                return image.get("image_url") or image.get("url")

            return (
                getattr(image, "image_url", None)
                or getattr(image, "image_path", None)
                or getattr(image, "url", None)
            )

    except (AttributeError, TypeError):
        pass

    return None


def _money(value):
    """Normalize a monetary value to two decimal places."""
    return Decimal(str(value)).quantize(Decimal("0.01"))


def _error(message, status_code):
    return {
        "success": False,
        "message": message,
    }, status_code

def create_order(user_id, data):
    """
    Create an order from the logged-in customer's cart.

    Required JSON:
    {
        "address_id": 1,
        "notes": "Optional delivery instructions"
    }
    """

    if not isinstance(data, dict):
        return _error("Invalid request body.", 400)

    address_id = data.get("address_id")

    if isinstance(address_id, bool) or not isinstance(address_id, int):
        return _error("address_id must be an integer.", 400)

    if address_id <= 0:
        return _error("address_id must be greater than zero.", 400)

    notes = data.get("notes")

    if notes is not None and not isinstance(notes, str):
        return _error("notes must be a string.", 400)

    if notes is not None:
        notes = notes.strip() or None

        if len(notes) > 5000:
            return _error("notes must not exceed 5000 characters.", 400)

    try:
        # Verify that the address belongs to the logged-in customer.
        address = db.session.execute(
            select(Address).where(
                Address.id == address_id,
                Address.user_id == user_id,
            )
        ).scalar_one_or_none()

        if address is None:
            db.session.rollback()
            return _error(
                "Delivery address not found for this customer.",
                404,
            )

        # Lock the cart row where supported by the database.
        cart = db.session.execute(
            select(Cart)
            .where(Cart.user_id == user_id)
            .with_for_update()
        ).scalar_one_or_none()

        if cart is None:
            db.session.rollback()
            return _error("Your cart is empty.", 400)

        # Lock cart items and load them.
        cart_items = db.session.execute(
            select(CartItem)
            .where(CartItem.cart_id == cart.id)
            .with_for_update()
        ).scalars().all()

        if not cart_items:
            db.session.rollback()
            return _error("Your cart is empty.", 400)

        # Validate every product before creating the order.
        order_lines = []
        subtotal = Decimal("0.00")

        for cart_item in cart_items:
            # Lock the product row where supported.
            product = db.session.execute(
                select(Product)
                .where(Product.id == cart_item.product_id)
                .with_for_update()
            ).scalar_one_or_none()

            if product is None:
                db.session.rollback()
                return _error(
                    f"Product {cart_item.product_id} no longer exists.",
                    400,
                )

            if not product.is_active:
                db.session.rollback()
                return _error(
                    f"Product '{product.product_name}' is unavailable.",
                    400,
                )

            if cart_item.quantity > product.stock_quantity:
                db.session.rollback()
                return {
                    "success": False,
                    "message": (
                        f"Insufficient stock for '{product.product_name}'."
                    ),
                    "product_id": product.id,
                    "requested_quantity": cart_item.quantity,
                    "available_stock": product.stock_quantity,
                }, 400

            unit_price = _money(_get_product_price(product))

            if unit_price < 0:
                db.session.rollback()
                return _error(
                    f"Invalid price for '{product.product_name}'.",
                    400,
                )

            line_total = _money(unit_price * cart_item.quantity)
            subtotal += line_total

            order_lines.append({
                "product": product,
                "quantity": cart_item.quantity,
                "unit_price": unit_price,
                "line_total": line_total,
                "product_name": product.product_name,
                "sku": product.sku,
                "image_url": _get_product_image(product),
            })

        subtotal = _money(subtotal)

        # Current model does not define shipping/tax calculation rules.
        discount_amount = Decimal("0.00")
        shipping_amount = Decimal("0.00")
        tax_amount = Decimal("0.00")

        total_amount = _money(
            subtotal
            - discount_amount
            + shipping_amount
            + tax_amount
        )

        # Create the order.
        order = Order(
            order_number=f"ORD-{uuid4().hex[:16].upper()}",
            user_id=user_id,
            address_id=address.id,
            subtotal=subtotal,
            discount_amount=discount_amount,
            shipping_amount=shipping_amount,
            tax_amount=tax_amount,
            total_amount=total_amount,
            payment_status="pending",
            order_status="placed",
            notes=notes,
        )

        db.session.add(order)
        db.session.flush()

        # Save product snapshots and deduct stock.
        for line in order_lines:
            product = line["product"]

            order_item = OrderItem(
                order_id=order.id,
                product_id=product.id,
                product_name=line["product_name"],
                product_image_url=line["image_url"],
                sku=line["sku"],
                quantity=line["quantity"],
                unit_price=line["unit_price"],
                discount_amount=Decimal("0.00"),
                total_price=line["line_total"],
            )

            db.session.add(order_item)

            product.stock_quantity -= line["quantity"]

        # Clear cart only after order creation logic succeeds.
        cart.items.clear()

        db.session.flush()

        # Build the response before committing.
        response_data = order.to_dict()

        db.session.commit()

        return {
            "success": True,
            "message": "Order created successfully.",
            "data": response_data,
        }, 201

    except SQLAlchemyError:
        db.session.rollback()

        return _error(
            "Unable to create order. Please try again.",
            500,
        )

    except (InvalidOperation, ValueError, TypeError):
        db.session.rollback()

        return _error(
            "Unable to calculate order prices.",
            400,
        )

def create_payment(user_id, data):
    try:
        order = db.session.execute(
            select(Order).where(
                Order.id == data["order_id"],
                Order.user_id == user_id
            )
        ).scalar_one_or_none()

        if order is None:
            return {
                "success": False,
                "message": "Order not found."
            }, 404

        if order.order_status in ("cancelled", "refunded"):
            return {
                "success": False,
                "message": "Payment is not allowed for this order."
            }, 400

        if order.payment_status == "success":
            return {
                "success": False,
                "message": "This order has already been paid."
            }, 409

        existing_payment = db.session.execute(
            select(Payment).where(
                Payment.order_id == order.id,
                Payment.payment_status == "pending"
            )
        ).scalar_one_or_none()

        if existing_payment is not None:
            return {
                "success": True,
                "message": "A pending payment already exists.",
                "payment": existing_payment.to_dict()
            }, 200

        amount = Decimal(str(order.total_amount))

        if amount < 0:
            return {
                "success": False,
                "message": "Invalid order amount."
            }, 400

        payment = Payment(
            order_id=order.id,
            payment_method=data["payment_method"],
            amount=amount,
            currency="INR",
            payment_status="pending"
        )

        db.session.add(payment)
        db.session.commit()

        return {
            "success": True,
            "message": "Payment initiated successfully. Awaiting payment confirmation.",
            "payment": payment.to_dict()
        }, 201

    except SQLAlchemyError:
        db.session.rollback()
        return {
            "success": False,
            "message": "Unable to initiate payment."
        }, 500

        for index, image in enumerate(image_files):

            if not image or not image.filename:
                continue

            original_name = secure_filename(image.filename)

            if "." not in original_name:
                raise ValueError("Invalid image filename.")

            extension = original_name.rsplit(".", 1)[1].lower()

            if extension not in ALLOWED_EXTENSIONS:
                raise ValueError(
                    f"Unsupported image format: {original_name}"
                )

            if (
                not image.mimetype
                or not image.mimetype.startswith("image/")
            ):
                raise ValueError(
                    f"Invalid image type: {original_name}"
                )

            image.stream.seek(0, os.SEEK_END)
            file_size = image.stream.tell()
            image.stream.seek(0)

            if file_size == 0:
                raise ValueError(
                    f"Image file is empty: {original_name}"
                )

            if file_size > MAX_IMAGE_SIZE:
                raise ValueError(
                    f"Image must be 5 MB or smaller: {original_name}"
                )

            filename = f"{uuid.uuid4().hex}.{extension}"

            file_path = os.path.join(
                upload_folder,
                filename
            )

            image.save(file_path)
            saved_files.append(file_path)

            image_record = ProductImage(
                product_id=product_id,
                image_url=(
                    f"/api/v1/vendor/products/uploads/{filename}"
                ),
                alt_text=os.path.splitext(original_name)[0][:250],
                is_primary=(index == 0),
                display_order=index
            )

            db.session.add(image_record)
            image_records.append(image_record)

        return {
            "success": True,
            "message": "Product images added successfully.",
            "data": [
                {
                    "image_url": record.image_url,
                    "alt_text": record.alt_text,
                    "is_primary": record.is_primary,
                    "display_order": record.display_order
                }
                for record in image_records
            ],
            "saved_files": saved_files
        }, 201

    except (ValueError, OSError) as error:

        for file_path in saved_files:
            try:
                if os.path.exists(file_path):
                    os.remove(file_path)
            except OSError:
                current_app.logger.exception(
                    "Failed to remove uploaded product image."
                )

        db.session.rollback()

        return {
            "success": False,
            "message": str(error),
            "saved_files": []
        }, 400

def create_product(data, vendor_id, commit=True):
    """Create a product after validating store ownership and category."""

    try:
        store_id = int(data["store_id"])
        category_id = int(data["category_id"])

        subcategory_id = (
            int(data["subcategory_id"])
            if data.get("subcategory_id") not in (None, "")
            else None
        )

        store = Store.query.filter_by(
            id=store_id,
            vendor_id=vendor_id
        ).first()

        if not store:
            return {
                "success": False,
                "message": "Store not found or access denied."
            }, 404

        if not store.is_active:
            return {
                "success": False,
                "message": "Cannot add products to an inactive store."
            }, 400

        category = db.session.get(Category, category_id)

        if not category or not category.is_active:
            return {
                "success": False,
                "message": "Category not found or inactive."
            }, 400

        if subcategory_id is not None:
            subcategory = db.session.get(
                Subcategory,
                subcategory_id
            )

            if (
                not subcategory
                or not subcategory.is_active
                or subcategory.category_id != category_id
            ):
                return {
                    "success": False,
                    "message": (
                        "Subcategory is invalid, inactive, "
                        "or does not belong to this category."
                    )
                }, 400

        product_name = data["product_name"].strip()
        slug = data["slug"].strip()
        description = data["description"].strip()
        sku = data["sku"].strip()

        if not all((product_name, slug, description, sku)):
            return {
                "success": False,

                    "Product name, slug, description, "
                    "and SKU cannot be empty."
                )
            }, 400

        price = Decimal(str(data["price"]))

        discount_price = (
            Decimal(str(data["discount_price"]))
            if data.get("discount_price") not in (None, "")
            else None
        )

        weight = (
            Decimal(str(data["weight"]))
            if data.get("weight") not in (None, "")
            else None
        )

        stock_value = data["stock_quantity"]

        if isinstance(stock_value, bool):
            raise ValueError

        stock_quantity = int(stock_value)

        if str(stock_quantity) != str(stock_value).strip():
            raise ValueError

        if not price.is_finite() or price <= 0:
            return {
                "success": False,
                "message": "Price must be greater than zero."
            }, 400

        if discount_price is not None:
            if (
                not discount_price.is_finite()
                or discount_price < 0
                or discount_price >= price
            ):
                return {
                    "success": False,
                    "message": (
                        "Discount price must be non-negative "
                        "and less than the original price."
                    )
                }, 400

        if weight is not None and (
            not weight.is_finite() or weight <= 0
        ):
            return {
                "success": False,
                "message": "Weight must be greater than zero."
            }, 400

        if stock_quantity < 0:
            return {
                "success": False,
                "message": "Stock quantity cannot be negative."
            }, 400

        featured_value = data.get("is_featured", False)

        if isinstance(featured_value, bool):
            is_featured = featured_value
        elif isinstance(featured_value, str) and (
            featured_value.lower() in ("true", "false", "1", "0")
        ):
            is_featured = featured_value.lower() in ("true", "1")
        else:
            return {
                "success": False,
                "message": "is_featured must be true or false."
            }, 400

        product = Product(
            store_id=store_id,
            category_id=category_id,
            subcategory_id=subcategory_id,
            product_name=product_name,
            slug=slug,
            description=description,

def create_review(user_id, data):
    if not isinstance(data, dict):
        return _error("A valid JSON object is required.", 400)

    product_id = data.get("product_id")
    rating = data.get("rating")
    title = data.get("title")
    comment = data.get("comment")
    order_item_id = data.get("order_item_id")

    # Validate product ID
    if isinstance(product_id, bool) or not isinstance(product_id, int):
        return _error("product_id must be an integer.", 400)

    if product_id <= 0:
        return _error("product_id must be greater than zero.", 400)

    # Validate rating
    if isinstance(rating, bool) or not isinstance(rating, int):
        return _error("rating must be an integer from 1 to 5.", 400)

    if rating < 1 or rating > 5:
        return _error("rating must be between 1 and 5.", 400)

    # Validate optional title
    if title is not None:
        if not isinstance(title, str):
            return _error("title must be text.", 400)

        title = title.strip() or None

        if title is not None and len(title) > 150:
            return _error("title must not exceed 150 characters.", 400)

    # Validate optional comment
    if comment is not None:
        if not isinstance(comment, str):
            return _error("comment must be text.", 400)

        comment = comment.strip() or None

    # Require an order item to verify the purchase.
    if isinstance(order_item_id, bool) or not isinstance(order_item_id, int):
        return _error(
            "order_item_id is required and must be an integer.",
            400,
        )

    if order_item_id <= 0:
        return _error("order_item_id must be greater than zero.", 400)

    try:
        product = db.session.get(Product, product_id)

        if product is None or not product.is_active:
            return _error("Active product not found.", 404)

        # Check that this order item belongs to the customer,
        # matches the product, and belongs to a delivered order.
        purchase = db.session.execute(
            select(OrderItem, Order)
            .join(Order, Order.id == OrderItem.order_id)
            .where(
                OrderItem.id == order_item_id,
                OrderItem.product_id == product_id,
                Order.user_id == user_id,
                func.lower(Order.order_status) == "delivered",
            )
        ).first()

        if purchase is None:
            return _error(
                "A delivered purchase of this product is required to review it.",
                403,
            )

        existing_review = db.session.execute(
            select(Review).where(
                Review.user_id == user_id,
                Review.product_id == product_id,
            )
        ).scalar_one_or_none()

        if existing_review is not None:
            return _error(
                "You have already reviewed this product.",
                409,
            )

        review = Review(
            user_id=user_id,
            product_id=product_id,
            order_item_id=order_item_id,
            rating=rating,
            title=title,
            comment=comment,
            is_verified_purchase=True,
            is_approved=False,
        )

        db.session.add(review)
        db.session.commit()

        return {
            "success": True,
            "message": "Review submitted successfully and is awaiting approval.",
            "data": review.to_dict(),
        }, 201

    except IntegrityError:
        db.session.rollback()
        return _error(
            "A review for this product already exists.",
            409,
        )

    except SQLAlchemyError:
        db.session.rollback()
        return _error("Unable to submit review.", 500)

def register_user(data):
    """
    Register a new user after checking email and phone uniqueness.
    """

    name = data["name"].strip()
    email = data["email"].strip().lower()
    phone_number = data["phone_number"].strip()
    password = data["password"]

    try:
        # Check whether email or phone number already exists.
        existing_user = db.session.execute(
            select(User).where(
                or_(
                    User.email == email,
                    User.phone_number == phone_number
                )
            )
        ).scalar_one_or_none()

        if existing_user:
            if existing_user.email == email:
                return {
                    "success": False,
                    "message": "Email is already registered."
                }, 409

            return {
                "success": False,
                "message": "Phone number is already registered."
            }, 409

        # Create a new user.
        user = User(
            name=name,
            email=email,
            phone_number=phone_number,
            password_hash=hash_password(password)
        )

        db.session.add(user)
        db.session.commit()

        return {
            "success": True,
            "message": "User registered successfully.",
            "user": user.to_dict()
        }, 201

    except IntegrityError:
        db.session.rollback()

        return {
            "success": False,
            "message": "Email or phone number is already registered."
        }, 409

    except SQLAlchemyError:
        db.session.rollback()

        return {
            "success": False,
            "message": "Unable to register user at this time."
        }, 500

def get_vendor_dashboard(vendor_id):
    try:
        # Vendor ke apne stores
        store_ids_query = select(Store.id).where(
            Store.vendor_id == vendor_id
        )

        total_stores = db.session.scalar(
            select(func.count()).select_from(Store).where(
                Store.vendor_id == vendor_id
            )
        ) or 0

        active_stores = db.session.scalar(
            select(func.count()).select_from(Store).where(
                Store.vendor_id == vendor_id,
                Store.is_active.is_(True)
            )
        ) or 0

        # Vendor ke apne products
        product_filter = Product.store_id.in_(store_ids_query)

        total_products = db.session.scalar(
            select(func.count()).select_from(Product).where(
                product_filter
            )
        ) or 0

        active_products = db.session.scalar(
            select(func.count()).select_from(Product).where(
                product_filter,
                Product.is_active.is_(True)
            )
        ) or 0

        total_stock = db.session.scalar(
            select(func.coalesce(func.sum(Product.stock_quantity), 0))
            .where(product_filter)
        ) or 0

        low_stock_products = db.session.scalar(
            select(func.count()).select_from(Product).where(
                product_filter,
                Product.stock_quantity <= 5
            )
        ) or 0

        # Sirf vendor ke products se related orders
        vendor_order_ids = (
            select(OrderItem.order_id)
            .join(Product, OrderItem.product_id == Product.id)
            .where(product_filter)
        )

        total_orders = db.session.scalar(
            select(func.count(distinct(OrderItem.order_id)))
            .select_from(OrderItem)
            .join(Product, OrderItem.product_id == Product.id)
            .where(product_filter)
        ) or 0

        pending_orders = db.session.scalar(
            select(func.count(distinct(OrderItem.order_id)))
            .select_from(OrderItem)
            .join(Product, OrderItem.product_id == Product.id)
            .join(Order, Order.id == OrderItem.order_id)

        ) or 0

        total_sales = db.session.scalar(

            )

            .join(Order, Order.id == OrderItem.order_id)
            .where(
                product_filter,
                Order.order_status.notin_(["cancelled", "refunded"])
            )
        ) or 0

            "message": "Vendor dashboard data fetched successfully.",

                "stores": {

                },

from decimal import Decimal

def _get_product_price(product):
    """Return the effective product price."""
    if product.discount_price is not None:
        return Decimal(str(product.discount_price))

    return Decimal(str(product.price))


def _get_product_image(product):
    """Return the first available product image URL."""
    try:
        images = product.images

        if images:
            image = images[0]

            if isinstance(image, dict):
                return image.get("image_url") or image.get("url")

            return (
                getattr(image, "image_url", None)
                or getattr(image, "image_path", None)
                or getattr(image, "url", None)
            )

    except (AttributeError, TypeError):
        pass

    return None


def _serialize_wishlist_item(item):
    """Serialize a wishlist item with product details."""
    product = item.product

    if product is None:
        return None

    price = Decimal(str(product.price))
    effective_price = _get_product_price(product)

    discount_percentage = 0

    if price > 0 and effective_price < price:
        discount_percentage = int(
            ((price - effective_price) / price) * 100
        )

    return {
        "wishlist_item_id": item.id,
        "product_id": product.id,
        "product_name": product.product_name,
        "slug": product.slug,
        "sku": product.sku,
        "description": product.description,
        "brand": product.brand,
        "price": str(price),
        "discount_price": (
            str(product.discount_price)
            if product.discount_price is not None
            else None
        ),
        "effective_price": str(effective_price),
        "discount_percentage": discount_percentage,
        "stock_quantity": product.stock_quantity,
        "is_active": product.is_active,
        "image_url": _get_product_image(product),
        "created_at": (
            item.created_at.isoformat()
            if item.created_at
            else None
        ),
    }


def _serialize_wishlist(items, user_id):
    """Build the wishlist response."""
    serialized_items = []

    for item in items:
        serialized_item = _serialize_wishlist_item(item)

        if serialized_item is not None:
            serialized_items.append(serialized_item)

    return {
        "user_id": user_id,
        "items": serialized_items,
        "item_count": len(serialized_items),
    }

                Wishlist.user_id == user_id,
            )
        ).scalar_one_or_none()

        if item is None:
            return {
                "success": False,
                "message": "Wishlist item not found.",
            }, 404

        db.session.delete(item)
        db.session.commit()

        return {
            "success": True,
            "message": "Product removed from wishlist successfully.",
            "wishlist_item_id": wishlist_item_id,
        }, 200

    except SQLAlchemyError:
        db.session.rollback()

        return {
            "success": False,
            "message": "Unable to remove wishlist item.",
        }, 500


def clear_wishlist(user_id):
    """Clear only the logged-in customer's wishlist."""
    try:
        items = db.session.execute(
            select(Wishlist).where(
                Wishlist.user_id == user_id
            )
        ).scalars().all()

        if not items:
            return {
                "success": True,
                "message": "Wishlist is already empty.",
                "wishlist": {
                    "user_id": user_id,
                    "items": [],
                    "item_count": 0,
                },
            }, 200

        for item in items:
            db.session.delete(item)

        db.session.commit()

        return {
            "success": True,
            "message": "Wishlist cleared successfully.",
            "wishlist": {
                "user_id": user_id,
                "items": [],
                "item_count": 0,
            },
        }, 200

    except SQLAlchemyError:
        db.session.rollback()

        return {
            "success": False,
            "message": "Unable to clear wishlist.",
        }, 500

    if user is None:
        return None, (

            404
        )

    if not user.is_active:
        return None, (

    return user, None

@inventory_bp.route("/inventory/movement", methods=["POST"])
@token_required
@role_required("admin", "vendor")
def create_inventory_movement():
    user, error = get_current_user()

    if error:
        return error

    data = request.get_json(silent=True)

    if not isinstance(data, dict):
        return jsonify({
            "success": False,
            "message": "Request body must be a valid JSON object."
        }), 400

    required_fields = {
        "product_id",
        "quantity_change",
        "movement_type"
    }

    missing_fields = required_fields - data.keys()

    if missing_fields:
        return jsonify({
            "success": False,
            "message": "Missing required fields.",
            "errors": {
                "missing_fields": sorted(missing_fields)
            }
        }), 400

    product_id = data["product_id"]
    quantity_change = data["quantity_change"]
    movement_type = data["movement_type"]

    # Validate product ID.
    if isinstance(product_id, bool) or not isinstance(
        product_id, (int, str)
    ):
        return jsonify({
            "success": False,
            "message": "Product ID must be an integer."
        }), 400

    # Validate quantity change.
    if isinstance(quantity_change, bool) or not isinstance(
        quantity_change, (int, str)
    ):
        return jsonify({
            "success": False,
            "message": "Quantity change must be an integer."
        }), 400

    try:
        product_id = int(product_id)
        quantity_change = int(quantity_change)

    except (TypeError, ValueError, OverflowError):
        return jsonify({
            "success": False,
            "message": (
                "Product ID and quantity change "
                "must be integers."
            )
        }), 400

    if product_id <= 0 or quantity_change == 0:
        return jsonify({
            "success": False,
            "message": (
                "Product ID must be positive and "
                "quantity change cannot be zero."
            )
        }), 400

    if not isinstance(movement_type, str):
        return jsonify({
            "success": False,
            "message": "Movement type must be a string."
        }), 400

    # Validate optional store ID.
    store_id = data.get("store_id")

    if store_id is not None:
        if isinstance(store_id, bool):
            return jsonify({
                "success": False,
                "message": "Invalid store ID."
            }), 400

        try:
            store_id = int(store_id)

        except (TypeError, ValueError, OverflowError):
            return jsonify({
                "success": False,
                "message": "Store ID must be an integer."
            }), 400

        if store_id <= 0:
            return jsonify({
                "success": False,
                "message": "Invalid store ID."
            }), 400

    # Vendor can modify inventory only for their own store.
    if user.role == "vendor":
        vendor_id = get_current_vendor_id(user)

        if vendor_id is None:
            return jsonify({
                "success": False,
                "message": "Vendor profile not found."
            }), 403

        if store_id is None:
            return jsonify({
                "success": False,
                "message": "store_id is required for vendors."
            }), 400

        store = Store.query.filter_by(
            id=store_id,
            vendor_id=vendor_id
        ).first()

        if store is None:
            return jsonify({
                "success": False,
                "message": "Store not found or access denied."
            }), 403

    # Record the movement and update stock through the operation.
    result, status_code = record_inventory_movement(
        product_id=product_id,
        quantity_change=quantity_change,
        movement_type=movement_type,
        store_id=store_id,
        reference_type=data.get("reference_type"),
        reference_id=data.get("reference_id"),
        note=data.get("note"),
        created_by_type=user.role,
        created_by_id=user.id,
    )

    return jsonify(result), status_code

# =========================================================
# Create Product + Upload Multiple Images
# =========================================================

@product_bp.route("", methods=["POST"])
@vendor_required
def add_product():
    """Create a product and upload its images in one request."""

    data = request.form.to_dict()
    image_files = request.files.getlist("images")

    if not data:
        return jsonify({
            "success": False,
            "message": "Product details are required as form data."
        }), 400

    image_files = [
        image for image in image_files
        if image and image.filename
    ]

    if not image_files:
        return jsonify({
            "success": False,
            "message": "At least one product image is required."
        }), 400

    result, status_code = create_product(
        data=data,
        vendor_id=request.vendor_id,
        commit=False
    )

    if status_code != 201:
        return jsonify(result), status_code

    product_id = result["data"]["id"]
    saved_files = []

    try:
        image_result, image_status = save_product_images(
            product_id=product_id,
            image_files=image_files
        )

        saved_files = image_result.get("saved_files", [])

        if image_status != 201 or not image_result.get("success"):
            db.session.rollback()

            for file_path in saved_files:
                try:
                    if os.path.isfile(file_path):
                        os.remove(file_path)
                except OSError:
                    current_app.logger.exception(
                        "Failed to clean up product image."
                    )

            return jsonify({
                "success": False,
                "message": image_result.get(
                    "message",
                    "Product image upload failed."
                )
            }), image_status

        # Save product and image records in one transaction.
        db.session.commit()

        product = db.session.get(Product, product_id)

        if product is None:
            return jsonify({
                "success": False,
                "message": "Product was saved but could not be retrieved."
            }), 500

        return jsonify({
            "success": True,
            "message": "Product and images created successfully.",
            "data": product.to_dict()
        }), 201

    except SQLAlchemyError:
        db.session.rollback()

        for file_path in saved_files:
            try:
                if os.path.isfile(file_path):
                    os.remove(file_path)
            except OSError:
                current_app.logger.exception(
                    "Failed to clean up product image."
                )

        return jsonify({
            "success": False,
            "message": "Database error while saving product images."
        }), 500

from decimal import Decimal, InvalidOperation


class ProductCreateSchema:

    REQUIRED_FIELDS = [
        "store_id",
        "category_id",
        "product_name",
        "slug",
        "description",
        "sku",
        "price",
        "stock_quantity",
    ]

    OPTIONAL_FIELDS = [
        "subcategory_id",
        "brand",
        "discount_price",
        "weight",
        "is_featured",
    ]

    @staticmethod
    def validate(data):
        errors = {}

        # Required fields
        for field in ProductCreateSchema.REQUIRED_FIELDS:
            value = data.get(field)

            if value is None or (
                isinstance(value, str) and not value.strip()
            ):
                errors[field] = f"{field} is required."

        # Stop if required fields are missing
        if errors:
            return False, errors

        # Integer fields
        for field in ("store_id", "category_id", "stock_quantity"):
            try:
                value = data[field]

                if isinstance(value, bool):
                    raise ValueError

                if isinstance(value, str) and not value.strip():
                    raise ValueError

                number = int(value)

                if str(number) != str(value).strip():
                    raise ValueError

                if field in ("store_id", "category_id") and number <= 0:
                    errors[field] = f"{field} must be greater than zero."

                elif field == "stock_quantity" and number < 0:
                    errors[field] = "Stock quantity cannot be negative."

            except (ValueError, TypeError):
                errors[field] = f"{field} must be a valid integer."

        # Optional integer field
        subcategory_id = data.get("subcategory_id")

        if subcategory_id not in (None, ""):
            try:
                if isinstance(subcategory_id, bool):
                    raise ValueError

                number = int(subcategory_id)

                if number <= 0 or str(number) != str(subcategory_id).strip():
                    raise ValueError

            except (ValueError, TypeError):
                errors["subcategory_id"] = (
                    "Subcategory ID must be a positive integer."
                )

        # Text fields
        for field in ("product_name", "slug", "description", "sku"):
            value = data.get(field)

            if isinstance(value, str):
                data[field] = value.strip()

                if not data[field]:
                    errors[field] = f"{field} cannot be empty."

        # Price validation
        for field in ("price", "discount_price", "weight"):
            value = data.get(field)

            if value in (None, ""):
                if field == "price":
                    errors[field] = "Price is required."
                else:
                    data[field] = None

                continue

            try:
                if isinstance(value, bool):
                    raise InvalidOperation

                number = Decimal(str(value))

                if not number.is_finite():
                    raise InvalidOperation

                if field == "price" and number <= 0:
                    errors[field] = "Price must be greater than zero."

                elif field == "discount_price" and number < 0:
                    errors[field] = "Discount price cannot be negative."

                elif field == "weight" and number <= 0:
                    errors[field] = "Weight must be greater than zero."

                elif field == "discount_price":
                    price_value = data.get("price")

                    try:
                        original_price = Decimal(str(price_value))

                        if original_price.is_finite() and number >= original_price:
                            errors[field] = (
                                "Discount price must be less than original price."
                            )

                    except (InvalidOperation, ValueError, TypeError):
                        pass

            except (InvalidOperation, ValueError, TypeError):
                errors[field] = f"{field} must be a valid number."

        # Optional brand
        brand = data.get("brand")

        if brand is not None and not isinstance(brand, str):
            errors["brand"] = "Brand must be text."
        elif isinstance(brand, str):
            data["brand"] = brand.strip() or None

        # Featured status
        is_featured = data.get("is_featured", False)

        if isinstance(is_featured, bool):
            data["is_featured"] = is_featured

        elif isinstance(is_featured, str):
            normalized = is_featured.strip().lower()

            if normalized in ("true", "1"):
                data["is_featured"] = True

            elif normalized in ("false", "0"):
                data["is_featured"] = False

            else:
                errors["is_featured"] = (
                    "is_featured must be true or false."
                )

        else:
            errors["is_featured"] = "is_featured must be true or false."

        return len(errors) == 0, errors


class ProductUpdateSchema:

    @staticmethod
    def validate(data):
        if not isinstance(data, dict) or not data:
            return False, {"body": "At least one field is required."}

        allowed_fields = {
            "category_id",
            "subcategory_id",
            "product_name",
            "slug",
            "description",
            "brand",
            "sku",
            "price",
            "discount_price",
            "stock_quantity",
            "weight",
            "is_featured",
        }

        errors = {}
        unknown_fields = set(data.keys()) - allowed_fields

        if unknown_fields:
            errors["fields"] = (
                "Unsupported fields: " + ", ".join(sorted(unknown_fields))
            )

        if errors:
            return False, errors

        # Validate supplied fields using the same rules as product creation.
        complete_data = {
            "store_id": "1",
            "category_id": data.get("category_id", "1"),
            "product_name": data.get("product_name", "Temporary Product"),
            "slug": data.get("slug", "temporary-product"),
            "description": data.get("description", "Temporary description"),
            "sku": data.get("sku", "TEMP-SKU"),
            "price": data.get("price", "1"),
            "stock_quantity": data.get("stock_quantity", "0"),
            "is_featured": data.get("is_featured", False),
        }

        for field in ProductCreateSchema.OPTIONAL_FIELDS:
            if field in data:
                complete_data[field] = data[field]

        valid, validation_errors = ProductCreateSchema.validate(complete_data)

        # Return only errors for fields the user actually supplied.
        relevant_errors = {
            field: message
            for field, message in validation_errors.items()
            if field in data
        }

        return valid and not relevant_errors, relevant_errors


class ProductStatusSchema:

    @staticmethod
    def validate(data):
        if not isinstance(data, dict):
            return False, {"body": "A JSON object is required."}

        if "is_active" not in data:
            return False, {"is_active": "is_active is required."}

        value = data["is_active"]

        if isinstance(value, bool):
            return True, {}

        if isinstance(value, str) and value.strip().lower() in (
            "true", "false", "1", "0"
        ):
            return True, {}

        return False, {
            "is_active": "is_active must be true or false."
        }

from flask import request


def paginate_query(query, default_page=1, default_per_page=20, max_per_page=100):
    try:
        page = max(1, request.args.get("page", default_page, type=int) or default_page)
        per_page = request.args.get("per_page", default_per_page, type=int) or default_per_page
    except (TypeError, ValueError):
        page = default_page
        per_page = default_per_page

    per_page = max(1, min(per_page, max_per_page))

    pagination = query.paginate(
        page=page,
        per_page=per_page,
        error_out=False
    )

    return {
        "data": pagination.items,
        "pagination": {
            "page": pagination.page,
            "per_page": pagination.per_page,
            "total": pagination.total,
            "pages": pagination.pages,
            "has_next": pagination.has_next,
            "has_prev": pagination.has_prev,
        }
    }

from flask import jsonify


def success_response(
    message="Request successful.",
    data=None,
    status_code=200,
    **extra
):
    body = {
        "success": True,
        "message": message,
    }

    if data is not None:
        body["data"] = data

    body.update(extra)

    return jsonify(body), status_code


def error_response(
    message="Request failed.",
    status_code=400,
    errors=None,
    **extra
):
    body = {
        "success": False,
        "message": message,
    }

    if errors is not None:
        body["errors"] = errors

    body.update(extra)

    return jsonify(body), status_code

from werkzeug.security import generate_password_hash, check_password_hash


def hash_password(password: str) -> str:
    """
    Convert a plain-text password into a secure password hash.
    """
    if not isinstance(password, str) or not password:
        raise ValueError("Password cannot be empty.")

    return generate_password_hash(password, method="scrypt")


def verify_password(password: str, password_hash: str) -> bool:
    """
    Verify a plain-text password against its stored hash.
    """
    if not isinstance(password, str) or not isinstance(password_hash, str):
        return False

    if not password or not password_hash:
        return False

    try:
        return check_password_hash(password_hash, password)
    except (ValueError, TypeError):
        return False

import re


def validate_required(value, field_name="Field"):
    if value is None or not str(value).strip():
        return False, f"{field_name} is required."
    return True, None


def validate_email(email):
    if not isinstance(email, str):
        return False, "A valid email address is required."

    email = email.strip()

    pattern = r"^[A-Za-z0-9._%+-]+@[A-Za-z0-9.-]+\.[A-Za-z]{2,}$"

    if not re.fullmatch(pattern, email):
        return False, "A valid email address is required."

    return True, None


def validate_positive_number(value, field_name="Value"):
    try:
        number = float(value)
    except (TypeError, ValueError):
        return False, f"{field_name} must be a valid number."

    if number <= 0:
        return False, f"{field_name} must be greater than zero."

    return True, None


def validate_phone(phone):
    if not isinstance(phone, str):
        return False, "A valid phone number is required."

    phone = phone.strip()

    if not re.fullmatch(r"\+?[0-9]{10,15}", phone):
        return False, "Phone number must contain 10 to 15 digits."

    return True, None
